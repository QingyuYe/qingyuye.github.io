# 1_batch_sampleprocess_v1.1.sh
# 关键提交脚本 依赖nohup和tasket 普适性更强 滚动提交
#!/bin/bash

# RNA-seq Analysis 滚动并发执行脚本 (High-Throughput) v1.1
# 功能：自动扫描 *.analysis.sh 并发执行，自动分配核心，记录脚本快照及执行命令
# author: Modified for zhouchi
# date: 2025/12/03

# 使用方法: nohup bash 1_batch_sampleprocess_v1.1.sh > 1_batch_runner.log 2>&1 &

  

# ================= 配置区域 =================

# 1. 设置最大并发任务数

max_jobs=10

# 2. 设置每个任务需要的核心数

threads_per_job=5


# 3. 自动生成 CPU 绑核列表 (无需手动填写)

# 逻辑：Slot 0 -> 0-4, Slot 1 -> 5-9, Slot 2 -> 10-14 ...

cpu_cores_list=()

for ((i=0; i<max_jobs; i++)); do

    start_core=$((i * threads_per_job))

    end_core=$((start_core + threads_per_job - 1))

    cpu_cores_list+=("${start_core}-${end_core}")

done

  

# 3. 指定包含 *.analysis.sh 的目录

work_dir=$(pwd)

  

# 4. 设置统一日志存放目录

log_dir="${work_dir}/1_Batch_Logs"

mkdir -p "$log_dir"

# ===========================================

  

# 状态追踪数组

declare -a pids=()

declare -a slot_ids=()

  

scripts=($(ls "${work_dir}"/*.analysis.sh 2>/dev/null))

  

if [ ${#scripts[@]} -eq 0 ]; then

    echo "❌ 错误：在 ${work_dir} 下未找到任何 .analysis.sh 脚本。"

    exit 1

fi

  

echo "📊 发现 ${#scripts[@]} 个任务脚本，准备开始滚动投递..."

echo "📂 日志目录: ${log_dir}"

echo "🚀 最大并发: ${max_jobs} (Total Cores Needed: $((max_jobs * threads_per_job)))"

echo "----------------------------------------"

  

for script_path in "${scripts[@]}"; do

    sample_name=$(basename "$script_path" .analysis.sh)

    # --- 核心逻辑：滚动等待空槽 ---

    while (( ${#pids[@]} >= max_jobs )); do

        for i in "${!pids[@]}"; do

            # 检查 PID 是否仍在运行

            if ! kill -0 "${pids[$i]}" 2>/dev/null; then

                echo "✅ 任务完成: PID ${pids[$i]} (Slot ${slot_ids[$i]})"

                unset 'pids[i]'

                unset 'slot_ids[i]'

            fi

        done

        # 重建数组索引 (去除空洞)

        pids=("${pids[@]}")

        slot_ids=("${slot_ids[@]}")

        # 如果仍然满载，等待 5 秒

        if (( ${#pids[@]} >= max_jobs )); then

            sleep 5

        fi

    done

    # ---------------------------

  

    # --- 分配 CPU 核心段 ---

    used_slots=("${slot_ids[@]}")

    free_slot=-1

    # 寻找最小的可用 Slot ID

    for idx in $(seq 0 $((max_jobs - 1))); do

        if [[ ! " ${used_slots[*]} " =~ " $idx " ]]; then

            free_slot=$idx

            break

        fi

    done

  

    # 安全兜底

    if [[ $free_slot -eq -1 ]]; then

        echo "⚠️ 警告：槽位分配异常，强制使用 Slot 0"

        free_slot=0

    fi

  

    cpu_range=${cpu_cores_list[$free_slot]}

    log_file="${log_dir}/${sample_name}.log"

  

    # 构建用于记录的命令字符串 (仅用于展示，加上转义符以便阅读)

    # 注意：这里我们构建一个字符串来展示实际上将要执行的命令

    record_cmd="nohup taskset -c \"${cpu_range}\" bash \"${script_path}\" >> \"${log_file}\" 2>&1 &"

  

    echo "▶️  启动样本: ${sample_name}"

    echo "    - 核心: $cpu_range (Slot $free_slot)"

    echo "    - 日志: $log_file"

  

    # ================= 记录日志与运行 =================

    # 1. 写入元数据、执行命令和代码快照 (使用覆盖模式 > )

    {

        echo "#########################################################"

        echo "# Task Meta Information"

        echo "# Sample: ${sample_name}"

        echo "# Start Time: $(date '+%Y-%m-%d %H:%M:%S')"

        echo "# Host: $(hostname)"

        echo "# Allocated CPU Cores: ${cpu_range} (Slot ${free_slot})"

        echo "# Script Path: ${script_path}"

        echo "# Actual Execution Command:"

        echo "# ${record_cmd}"

        echo "#########################################################"

        echo ""

        echo "⬇️ ⬇️ ⬇️  Snapshot of the Script Code  ⬇️ ⬇️ ⬇️"

        echo "---------------------------------------------------------"

        cat "$script_path"

        echo ""

        echo "---------------------------------------------------------"

        echo "⬆️ ⬆️ ⬆️  End of Script Code  ⬆️ ⬆️ ⬆️"

        echo ""

        echo "################## Execution Output #####################"

        echo ""

    } > "$log_file"

  

    # 2. 启动任务 (使用追加模式 >> )

    # 这里执行实际的命令

    nohup taskset -c "$cpu_range" bash "$script_path" >> "$log_file" 2>&1 &

    new_pid=$!

    pids+=($new_pid)

    slot_ids+=($free_slot)

    # 稍微暂停防止瞬间 IO 拥堵

    sleep 1

  

done

  

echo "----------------------------------------"

echo "⏳ 所有任务已投递，等待剩余任务完成..."

for pid in "${pids[@]}"; do

    wait "$pid"

done

  

echo "🎉 所有样本处理完成！请检查目录: ${log_dir}"

# 完成第一步后生成报告

#!/usr/bin/env python3

# -*- coding: utf-8 -*-

# RNA-seq Analysis Report Generator

# Author: Modified for zhouchi

# Date: 2025/12/03

  

import os

import re

import json

import csv

import glob

import argparse

from datetime import datetime

  

# ================= 配置区域 =================

# 默认的工作根目录

DEFAULT_PIPELINE_DIR = "/BDproject2/User/zhouchi/15_jiechang/2_BulkRNApipeline"

DEFAULT_LOG_DIR = "./1_Batch_Logs"

# ===========================================

  

def parse_fastp_json(json_path):

    """解析 Fastp JSON 获取质控数据"""

    try:

        with open(json_path, 'r') as f:

            data = json.load(f)

            summary = data.get('summary', {})

            before = summary.get('before_filtering', {})

            after = summary.get('after_filtering', {})

            return {

                'raw_reads': before.get('total_reads', 0),

                'clean_reads': after.get('total_reads', 0),

                'q30_rate': after.get('q30_rate', 0),

                'adapter_trimmed_reads': data.get('adapter_cutting', {}).get('adapter_trimmed_reads', 0)

            }

    except Exception as e:

        return None

  

def parse_hisat_logs(log_path):

    """

    从 Runner 生成的总 Log 文件中提取三次 Hisat2 的比对率

    兼容标准格式和 --new-summary 格式

    """

    rates = []

    try:

        with open(log_path, 'r') as f:

            content = f.read()

            # 优先尝试匹配 --new-summary 格式

            matches = re.findall(r"Overall alignment rate:\s+(\d+\.?\d*)%", content)

            # 如果没找到，尝试匹配旧格式

            if not matches:

                matches = re.findall(r"(\d+\.?\d*)% overall alignment rate", content)

            rates = [float(m) for m in matches]

    except Exception as e:

        print(f"Warning parsing log {log_path}: {e}")

        pass

    result = {'mycop_rate': 'N/A', 'rrna_rate': 'N/A', 'genome_map_rate': 'N/A'}

    if len(rates) >= 1: result['mycop_rate'] = rates[0]

    if len(rates) >= 2: result['rrna_rate'] = rates[1]

    if len(rates) >= 3: result['genome_map_rate'] = rates[2]

    return result

  

def parse_featurecounts(summary_path):

    """解析 featureCounts .summary 文件"""

    assigned = 0

    total = 0

    try:

        with open(summary_path, 'r') as f:

            lines = f.readlines()

            for line in lines:

                parts = line.strip().split('\t')

                if len(parts) < 2: continue

                if parts[0] == 'Assigned':

                    assigned = int(parts[1])

                if parts[0] != 'Status':

                    total += int(parts[1])

    except Exception as e:

        return None

    return {

        'fc_assigned': assigned,

        'fc_rate': (assigned / total * 100) if total > 0 else 0

    }

  

def main():

    parser = argparse.ArgumentParser(description="Generate RNA-seq QC Report")

    parser.add_argument('--dir', default=DEFAULT_PIPELINE_DIR, help="Pipeline output directory")

    parser.add_argument('--logs', default=DEFAULT_LOG_DIR, help="Batch runner logs directory")

    parser.add_argument('--out', default="1_RNAseq_Summary_Report", help="Output filename prefix")

    args = parser.parse_args()

  

    json_pattern = os.path.join(args.dir, "trim_galore", "*.result.json")

    json_files = glob.glob(json_pattern)

    if not json_files:

        print(f"❌ Error: No fastp JSON files found in {os.path.join(args.dir, 'trim_galore')}")

        return

  

    results = []

    print(f"📊 Found {len(json_files)} samples. Parsing data...")

  

    for json_file in sorted(json_files):

        filename = os.path.basename(json_file)

        sample = filename.replace(".result.json", "")

        print(f"   -> Processing {sample}...")

        fastp_data = parse_fastp_json(json_file)

        log_file = os.path.join(args.logs, f"{sample}.log")

        if not os.path.exists(log_file):

             print(f"      ⚠️ Warning: Log file not found: {log_file}")

             hisat_data = {}

        else:

             hisat_data = parse_hisat_logs(log_file)

        fc_file = os.path.join(args.dir, "featurecounts", f"{sample}.read.count.summary")

        fc_data = parse_featurecounts(fc_file)

        row = {'Sample': sample}

        if fastp_data: row.update(fastp_data)

        if hisat_data: row.update(hisat_data)

        if fc_data: row.update(fc_data)

        results.append(row)

  

    # ================= 输出 CSV =================

    csv_file = args.out + ".csv"

    headers = [

        'Sample', 'raw_reads', 'clean_reads', 'adapter_trimmed_reads', 'q30_rate',

        'mycop_rate', 'rrna_rate', 'genome_map_rate',

        'fc_assigned', 'fc_rate'

    ]

    with open(csv_file, 'w', newline='') as f:

        writer = csv.DictWriter(f, fieldnames=headers, extrasaction='ignore')

        writer.writeheader()

        for row in results:

            clean_row = row.copy()

            # [FIX] 专门处理 q30_rate，如果是小数则乘以 100

            if 'q30_rate' in clean_row:

                val = clean_row['q30_rate']

                if isinstance(val, (int, float)) and val <= 1.0:

                    clean_row['q30_rate'] = val * 100

  

            # 格式化所有浮点数

            for k, v in clean_row.items():

                if isinstance(v, float):

                    clean_row[k] = f"{v:.2f}"

            writer.writerow(clean_row)

    print(f"✅ CSV Report generated: {csv_file}")

  

    # ================= 输出 简单 HTML =================

    html_file = args.out + ".html"

    with open(html_file, 'w') as f:

        f.write('''<html><head><style>

            table {border-collapse: collapse; width: 100%; font-family: Arial;}

            th, td {padding: 8px; text-align: center; border-bottom: 1px solid #ddd;}

            th {background-color: #f2f2f2;}

            tr:hover {background-color: #f5f5f5;}

            .good {color: green; font-weight: bold;}

            .warn {color: orange; font-weight: bold;}

            .bad {color: red; font-weight: bold;}

        </style></head><body><h2>RNA-seq Pipeline Summary</h2><table>''')

        f.write("<tr>")

        for h in headers: f.write(f"<th>{h}</th>")

        f.write("</tr>")

        for row in results:

            f.write("<tr>")

            f.write(f"<td>{row.get('Sample')}</td>")

            f.write(f"<td>{row.get('raw_reads', 0):,}</td>")

            f.write(f"<td>{row.get('clean_reads', 0):,}</td>")

            f.write(f"<td>{row.get('adapter_trimmed_reads', 0):,}</td>")

            # Q30 Color

            q30 = row.get('q30_rate', 0)

            if q30 <= 1.0: q30 *= 100

            cls = "good" if q30 > 90 else ("warn" if q30 > 80 else "bad")

            f.write(f"<td class='{cls}'>{q30:.2f}%</td>")

            # Mycoplasma

            myco = row.get('mycop_rate')

            if isinstance(myco, (int, float)):

                cls = "good" if myco < 1 else ("warn" if myco < 5 else "bad")

                f.write(f"<td class='{cls}'>{myco}%</td>")

            else: f.write(f"<td>{myco}</td>")

  

            # rRNA

            rrna = row.get('rrna_rate')

            if isinstance(rrna, (int, float)):

                cls = "good" if rrna < 10 else "warn"

                f.write(f"<td class='{cls}'>{rrna}%</td>")

            else: f.write(f"<td>{rrna}</td>")

            # Genome Map

            gmap = row.get('genome_map_rate')

            if isinstance(gmap, (int, float)):

                cls = "good" if gmap > 80 else ("warn" if gmap > 60 else "bad")

                f.write(f"<td class='{cls}'>{gmap}%</td>")

            else: f.write(f"<td>{gmap}</td>")

  

            # FeatureCounts

            f.write(f"<td>{row.get('fc_assigned', 0):,}</td>")

            fc_rate = row.get('fc_rate', 0)

            cls = "good" if fc_rate > 60 else ("warn" if fc_rate > 40 else "bad")

            f.write(f"<td class='{cls}'>{fc_rate:.2f}%</td>")

            f.write("</tr>")

        f.write("</table></body></html>")

    print(f"✅ HTML Report generated: {html_file}")

  

if __name__ == "__main__":

    main()
