scan_md_fences <- function(lines) {
  mask <- rep(FALSE, length(lines))
  open_at <- NA_integer_
  marker_char <- ""
  marker_width <- 0L

  for (i in seq_along(lines)) {
    x <- lines[i]

    if (is.na(open_at)) {
      m <- regmatches(
        x,
        regexec("^ {0,3}(`{3,}|~{3,})(.*)$", x, perl = TRUE)
      )[[1]]

      if (length(m) > 0L) {
        open_at <- i
        marker_char <- substr(m[2], 1L, 1L)
        marker_width <- nchar(m[2])
        mask[i] <- TRUE
      }

    } else {
      mask[i] <- TRUE

      m <- regmatches(
        x,
        regexec("^ {0,3}(`{3,}|~{3,})[ \t]*$", x, perl = TRUE)
      )[[1]]

      if (length(m) > 0L &&
          substr(m[2], 1L, 1L) == marker_char &&
          nchar(m[2]) >= marker_width) {
        open_at <- NA_integer_
        marker_char <- ""
        marker_width <- 0L
      }
    }
  }

  list(mask = mask, unclosed = open_at)
}


check_markdown <- function(md_file) {

  md_file <- normalizePath(md_file, winslash = "/", mustWork = TRUE)
  lines <- readLines(md_file, encoding = "UTF-8", warn = FALSE)

  problems <- list()

  add_issue <- function(level, line, detail) {
    problems[[length(problems) + 1L]] <<- data.frame(
      level = level,
      line = as.integer(line),
      detail = detail,
      stringsAsFactors = FALSE
    )
  }

  cat("========================================\n")
  cat("Markdown 格式检查\n")
  cat("文件：", basename(md_file), "\n")
  cat("========================================\n\n")

  if (!grepl(
    "^[0-9]{4}-[0-9]{2}-[0-9]{2}-.+\\.md$",
    basename(md_file)
  )) {
    add_issue("ERROR", NA_integer_,
              "文件名不符合 YYYY-MM-DD-标题.md 格式")
  }

  if (!length(lines)) {
    add_issue("ERROR", 1L, "Markdown 文件为空")
  } else {

    first <- sub("^\ufeff", "", lines[1])

    if (!identical(first, "---")) {
      add_issue("ERROR", 1L,
                "第一行必须是 ---，不能有空行或其他正文")
    } else {

      yaml_end <- which(
        seq_along(lines) > 1L &
        trimws(lines) == "---"
      )

      if (!length(yaml_end)) {
        add_issue("ERROR", 1L, "YAML 属性区没有结束标记 ---")
      } else {

        yaml_end <- yaml_end[1]

        yaml_lines <- if (yaml_end > 2L) {
          lines[2:(yaml_end - 1L)]
        } else {
          character(0)
        }

        yaml_data <- tryCatch(
          yaml::yaml.load(
            paste(yaml_lines, collapse = "\n"),
            eval.expr = FALSE
          ),
          error = function(e) e
        )

        if (inherits(yaml_data, "error")) {
          add_issue(
            "ERROR", 2L,
            paste0("YAML 解析失败：", conditionMessage(yaml_data))
          )
        } else if (!is.list(yaml_data) ||
                   is.null(names(yaml_data))) {
          add_issue("ERROR", 2L,
                    "YAML 属性区不是有效的键值映射")
        } else {

          for (key in c("title", "date", "type", "summary", "tags")) {
            if (is.null(yaml_data[[key]])) {
              add_issue(
                "WARNING", 2L,
                paste0("缺少文章属性：", key)
              )
            }
          }

          if (!is.null(yaml_data$published) &&
              identical(yaml_data$published, FALSE)) {
            add_issue("WARNING", 2L,
                      "published: false 会使文章不发布")
          }
        }

        # 检查正文开头是否又出现了一组 YAML
        next_line <- yaml_end + 1L

        while (next_line <= length(lines) &&
               !nzchar(trimws(lines[next_line]))) {
          next_line <- next_line + 1L
        }

        if (next_line <= length(lines) &&
            trimws(lines[next_line]) == "---") {

          probe_end <- min(next_line + 12L, length(lines))

          probe <- if (next_line < probe_end) {
            lines[(next_line + 1L):probe_end]
          } else {
            character(0)
          }

          if (any(grepl(
            "^[[:space:]]*(title|date|type|summary|tags):",
            probe
          ))) {
            add_issue(
              "ERROR", next_line,
              "疑似重复 YAML 属性区：Obsidian 可能已有 Properties，正文又粘贴了一组属性"
            )
          }
        }
      }
    }

    # 检查代码围栏
    fences <- scan_md_fences(lines)

    if (!is.na(fences$unclosed)) {
      add_issue(
        "ERROR",
        fences$unclosed,
        "代码块没有正确关闭，请检查 ``` 或 ~~~"
      )
    }

    # 检查 Obsidian 图片引用
    wiki_pattern <- "!\\[\\[([^]\\r\\n]+)\\]\\]"
    active_lines <- lines[!fences$mask]

    matches <- regmatches(
      active_lines,
      gregexpr(wiki_pattern, active_lines, perl = TRUE)
    )

    refs <- unlist(matches, use.names = FALSE)

    if (length(refs)) {

      get_name <- function(x) {
        x <- sub("^!\\[\\[", "", x)
        x <- sub("\\]\\]$", "", x)
        x <- strsplit(x, "|", fixed = TRUE)[[1]][1]
        basename(gsub("\\\\", "/", trimws(x)))
      }

      names <- unique(vapply(refs, get_name, character(1)))

      image_names <- names[grepl(
        "\\.(png|jpe?g|gif|webp|svg|bmp|tiff?)$",
        names,
        ignore.case = TRUE
      )]

      att_dir <- file.path(dirname(md_file), "attachments")

      local_images <- if (dir.exists(att_dir)) {
        basename(list.files(
          att_dir,
          full.names = TRUE,
          recursive = TRUE,
          pattern = "\\.(png|jpe?g|gif|webp|svg|bmp|tiff?)$",
          ignore.case = TRUE
        ))
      } else {
        character(0)
      }

      missing_images <- setdiff(image_names, local_images)

      if (length(missing_images)) {
        for (name in missing_images) {
          add_issue(
            "WARNING",
            NA_integer_,
            paste0(
              "本地 attachments 中找不到图片：", name,
              "；如果已上传到 GitHub，可忽略"
            )
          )
        }
      }
    }
  }

  issues <- if (length(problems)) {
    do.call(rbind, problems)
  } else {
    data.frame(
      level = character(0),
      line = integer(0),
      detail = character(0)
    )
  }

  ok <- !any(issues$level == "ERROR")

  if (nrow(issues)) {
    for (i in seq_len(nrow(issues))) {
      location <- if (is.na(issues$line[i])) {
        "文件"
      } else {
        paste0("第 ", issues$line[i], " 行")
      }

      cat(
        sprintf("[%s] %s：%s\n",
                issues$level[i],
                location,
                issues$detail[i])
      )
    }
  }

  cat("\n检查结果：",
      if (ok) "PASS（无阻断性错误）" else "FAIL（存在错误）",
      "\n")

  cat("总行数：", length(lines), "\n")
  cat("错误数量：", sum(issues$level == "ERROR"), "\n")
  cat("警告数量：", sum(issues$level == "WARNING"), "\n")

  invisible(list(
    ok = ok,
    issues = issues,
    md_file = md_file
  ))
}

# ============================================================
# Obsidian Markdown 软换行转硬换行
# 普通正文自动添加行尾两个空格
# 跳过 YAML、代码块、标题、列表、表格
# ============================================================

fix_md_linebreaks <- function(lines) {

  n <- length(lines)
  if (n < 2L) return(lines)

  # 1. 跳过代码围栏及其内部内容
  skip <- scan_md_fences(lines)$mask

  # 2. 跳过 YAML Front Matter
  if (sub("^\ufeff", "", lines[1]) == "---") {
    yaml_end <- which(
      seq_along(lines) > 1L &
      trimws(lines) == "---"
    )

    if (length(yaml_end)) {
      skip[seq_len(yaml_end[1])] <- TRUE
    }
  }

  # 3. 识别需要保护的 Markdown 结构
  is_special <- function(x) {

    grepl(
      "^([ \t]{4,}| {0,3}(#{1,6}[ \t]+|>|[-+*][ \t]+|[0-9]+[.)][ \t]+|\\|)|[ \t]*<)",
      x,
      perl = TRUE
    ) ||
      grepl("|", x, fixed = TRUE)
  }

  # 4. 只处理普通正文中的软换行
  for (i in seq_len(n - 1L)) {

    current <- lines[i]
    following <- lines[i + 1L]

    # 跳过 YAML 和代码块
    if (skip[i] || skip[i + 1L]) next

    # 跳过空行
    if (!nzchar(trimws(current))) next
    if (!nzchar(trimws(following))) next

    # 跳过标题、列表、表格等结构
    if (is_special(current) || is_special(following)) next

    # 已经有 Markdown 硬换行则不重复添加
    if (grepl(" {2,}$", current)) next
    if (grepl("\\\\$", current)) next

    # 在行末添加两个空格
    lines[i] <- paste0(
      sub("[ \t]+$", "", current),
      "  "
    )
  }

  return(lines)
}

publish_obsidian <- function(md_file,
                            owner = "QingyuYe",
                            repo = "qingyuye.github.io",
                            branch = "master",
                            overwrite_images = FALSE) {

  token <- Sys.getenv("GITHUB_PAT")
  if (!nzchar(token)) stop("GitHub Token 未加载，请先运行 Cell 1。")
  if (!exists("scan_md_fences", mode = "function")) {
    stop("请先运行 Cell 2，定义 scan_md_fences()。")
  }

  md_file <- normalizePath(md_file, winslash = "/", mustWork = TRUE)
  md_name <- basename(md_file)
  lines <- readLines(md_file, encoding = "UTF-8", warn = FALSE)
  if (!length(lines)) stop("Markdown 文件为空。")
  lines[1] <- sub("^\ufeff", "", lines[1])

  att_dir <- file.path(dirname(md_file), "attachments")
  img_pattern <- "\\.(png|jpe?g|gif|webp|svg|bmp|tiff?)$"

  img_files <- if (dir.exists(att_dir)) {
    list.files(
      att_dir,
      pattern = img_pattern,
      full.names = TRUE,
      recursive = TRUE,
      ignore.case = TRUE
    )
  } else {
    character(0)
  }

  if (anyDuplicated(tolower(basename(img_files)))) {
    stop("attachments 内存在重名图片，请先处理。")
  }

  # ==========================================
  # 1. 转换 Obsidian 图片引用
  # ==========================================

  fences <- scan_md_fences(lines)
  skip_mask <- fences$mask

  if (identical(trimws(lines[1]), "---")) {
    yaml_end <- which(
      seq_along(lines) > 1L & trimws(lines) == "---"
    )
    if (length(yaml_end)) {
      skip_mask[seq_len(yaml_end[1])] <- TRUE
    }
  }

  target_lines <- which(!skip_mask)
  body_lines <- lines[target_lines]

  pattern <- "!\\[\\[([^\\r\\n]+?)\\]\\]"
  matches <- gregexpr(pattern, body_lines, perl = TRUE)
  found <- regmatches(body_lines, matches)

  get_filename <- function(x) {
    x <- substr(x, 4L, nchar(x) - 2L)
    x <- strsplit(x, "|", fixed = TRUE)[[1]][1]
    basename(gsub("\\\\", "/", trimws(x)))
  }

  is_image <- function(x) {
    grepl(img_pattern, get_filename(x), ignore.case = TRUE)
  }

  convert_image <- function(x) {
    if (!is_image(x)) return(x)

    filename <- get_filename(x)
    alt <- tools::file_path_sans_ext(filename)
    alt <- gsub("[\\[\\]]", "", alt)

    url <- utils::URLencode(
      enc2utf8(filename),
      reserved = TRUE
    )

    paste0("![", alt, "](/images/", url, ")")
  }

  raw_refs <- unlist(found, use.names = FALSE)

  image_refs <- if (length(raw_refs)) {
    selected <- raw_refs[vapply(
      raw_refs, is_image, logical(1)
    )]
    unique(vapply(selected, get_filename, character(1)))
  } else {
    character(0)
  }

  converted_count <- if (length(raw_refs)) {
    sum(vapply(raw_refs, is_image, logical(1)))
  } else {
    0L
  }

  if (length(body_lines) && length(raw_refs)) {
    replacements <- lapply(
      found,
      function(x) vapply(x, convert_image, character(1))
    )
    regmatches(body_lines, matches) <- replacements
    lines[target_lines] <- body_lines
  }

  # ==========================================
  # 2. GitHub API
  # ==========================================

  api_url <- function(path) {
    parts <- strsplit(path, "/", fixed = TRUE)[[1]]

    encoded <- vapply(parts, function(x) {
      utils::URLencode(enc2utf8(x), reserved = TRUE)
    }, character(1))

    paste0(
      "https://api.github.com/repos/",
      owner, "/", repo, "/contents/",
      paste(encoded, collapse = "/")
    )
  }

  make_request <- function(path) {
    httr2::request(api_url(path)) |>
      httr2::req_auth_bearer_token(token) |>
      httr2::req_headers(
        Accept = "application/vnd.github+json",
        `X-GitHub-Api-Version` = "2022-11-28",
        `User-Agent` = "Obsidian-R-Publisher"
      ) |>
      httr2::req_error(
        is_error = function(resp) FALSE
      )
  }

  check_response <- function(resp, path, valid_status) {
    status <- httr2::resp_status(resp)

    if (status %in% valid_status) {
      return(invisible(TRUE))
    }

    detail <- tryCatch(
      httr2::resp_body_json(resp)$message,
      error = function(e) "Unknown API error"
    )

    stop(
      "GitHub API ", status, "：", path,
      "\n", detail
    )
  }

  get_sha <- function(path) {
    resp <- make_request(path) |>
      httr2::req_url_query(ref = branch) |>
      httr2::req_perform()

    if (httr2::resp_status(resp) == 404L) {
      return(NULL)
    }

    check_response(resp, path, 200L)
    httr2::resp_body_json(resp)$sha
  }

  upload <- function(path, encoded, sha = NULL) {
    body <- list(
      message = paste("Publish", path),
      content = encoded,
      branch = branch
    )

    if (!is.null(sha)) body$sha <- sha

    resp <- make_request(path) |>
      httr2::req_method("PUT") |>
      httr2::req_body_json(body, auto_unbox = TRUE) |>
      httr2::req_perform()

    check_response(resp, path, c(200L, 201L))
    message("上传成功：", path)
  }

  # ==========================================
  # 3. 上传 attachments 图片
  # ==========================================

  message("开始处理图片……")

  uploaded_count <- 0L
  skipped_count <- 0L

  for (src in img_files) {
    dest <- paste0("images/", basename(src))
    sha <- get_sha(dest)

    if (!is.null(sha) && !overwrite_images) {
      message("已存在，跳过：", dest)
      skipped_count <- skipped_count + 1L
      next
    }

    encoded <- base64enc::base64encode(
      src, linewidth = 0
    )

    upload(dest, encoded, sha)
    uploaded_count <- uploaded_count + 1L
  }

  # ==========================================
  # 4. 检查 Markdown 引用的图片
  # ==========================================

  for (filename in image_refs) {
    dest <- paste0("images/", filename)

    if (is.null(get_sha(dest))) {
      stop(
        "图片不存在：", filename,
        "\n请放入 attachments，或先上传到 GitHub/images。"
      )
    }
  }

# ==========================================
# 5. 修复软换行并上传 Markdown
# ==========================================

# 只修改内存中的 Markdown，不修改 Obsidian 原文件
  lines <- fix_md_linebreaks(lines)

  md_path <- paste0("_posts/", md_name)

  md_text <- paste0(
    paste(lines, collapse = "\n"),
    "\n"
  )

  encoded_md <- base64enc::base64encode(
    charToRaw(enc2utf8(md_text)),
    linewidth = 0
  )

  upload(md_path, encoded_md, get_sha(md_path))

  # ==========================================
  # 6. 输出结果
  # ==========================================

  message("\n========================================")
  message("发布完成")
  message("========================================")
  message("文章：", md_name)
  message("转换图片引用：", converted_count)
  message("附件图片总数：", length(img_files))
  message("上传/覆盖图片：", uploaded_count)
  message("跳过已有图片：", skipped_count)
  message("分支：", branch)
  message("仓库：https://github.com/", owner, "/", repo)

  invisible(list(
    article = md_path,
    converted_images = converted_count,
    uploaded_images = uploaded_count,
    skipped_images = skipped_count
  ))
}
